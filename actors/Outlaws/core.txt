/*Reserved frames

Standing = A
Walking = B-E
Running = F-I
Shooting = J-N
Pain = O
Melee = P-R
Dynamite = S-T
Crouching = U-V
StrafeRun = W-X
StrafeWalk = Y-Z

Death Animations typically have 5 for the first 5, and 6 each for the last 2.
We typically put those in a separate sprite name.

*/

//Generic variables used by this controller
Actor Outlaws_Enemy_AllowWalk : TemporaryBoolean {}
Actor Outlaws_Enemy_AllowRun : TemporaryBoolean {}
Actor Outlaws_Enemy_AllowMelee : TemporaryBoolean {}
Actor Outlaws_Enemy_AllowStrafe : TemporaryBoolean {}
Actor Outlaws_Enemy_ShootType : TemporaryCounter {}
Actor Outlaws_Enemy_AllowDynamite : TemporaryBoolean {}
Actor Outlaws_Enemy_CrouchType : TemporaryCounter {}
Actor Outlaws_Enemy_Difficulty : TemporaryCounter {}
Actor Outlaws_Enemy_ID : TemporaryCounter {}

//Controller definition
Actor OutlawsEnemyCore : SamsaraMonsterHumanoid
{
	Height 56
	Radius 20
	Health 5
	Mass 30
	XScale 0.67
	YScale 0.55833
	Speed 6
	Monster
	Painchance 128
	States
	{
		//Idle/standing
		Spawn:
			"####" A 4 A_Look
			Wait
			
		//Movement based states
		See:
			"####" A 0 A_JumpIfInventory("Outlaws_Enemy_AllowWalk",1,"AllowWalk")
			"####" A 0 A_JumpIfInventory("Outlaws_Enemy_AllowRun",1,"Run")
			Goto StandingEnemy
		StandingEnemy:
			"####" A 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" AAAA 4 A_Chase("Melee","Missile",CHF_DONTMOVE)
			Loop
		AllowWalk:
			"####" "#" 0 A_Jump(64,"Walk")
		Run:
			"####" F 0 ACS_NamedExecuteWithResult("Outlaws_EnemySpeed",12)
			"####" F 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" F 4 A_Chase
			"####" G 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" G 4 A_Chase
			"####" H 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" H 4 A_Chase
			"####" I 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" I 4 A_Chase
			Goto See
		Walk:
			"####" F 0 ACS_NamedExecuteWithResult("Outlaws_EnemySpeed",6)
			"####" B 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" B 4 A_Chase
			"####" C 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" C 4 A_Chase
			"####" D 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" D 4 A_Chase
			"####" E 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" E 4 A_Chase
			Goto See
			
		//Attack based states
		Missile:
			"####" "#" 0 A_Jump(224,2)
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowDynamite",1,"Dynamite")
			"####" "#" 0 A_Jump(224,2)
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowStrafe",1,"DoStrafe")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",2,"Fire5Frames")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",1,"Fire3Frames")
		Fire3Frames:
			"####" JKL 4 A_FaceTarget
			"####" N 0 //Do attack stuff
			Goto See
		Fire5Frames:
			"####" JKLM 4 A_FaceTarget
			"####" N 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			Goto See
		Melee:
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowMelee",1,1)
			Goto Missile
			"####" PQR 4 A_FaceTarget
			"####" R 4 Bright //Do attack stuff
			Goto See
		Dynamite:
			"####" S 12 Bright //Do attack stuff
			Goto See
		DoStrafe:
			"####" "#" 0 A_Jump(256,"StrafeRun","StrafeWalk")
		StrafeRun:
			"####" W 4 A_ChangeVelocity(0,2,0,CVF_RELATIVE|CVF_REPLACE)
			"####" X 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" X 4 Bright A_ChangeVelocity(0,2,0,CVF_RELATIVE|CVF_REPLACE)
			Goto See
		StrafeWalk:
			"####" Y 4 A_ChangeVelocity(0,1,0,CVF_RELATIVE|CVF_REPLACE)
			"####" Z 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" Z 4 Bright A_ChangeVelocity(0,1,0,CVF_RELATIVE|CVF_REPLACE)
			Goto See
			
		//Pain state
		Pain:
			"####" "#" 0 A_Jump(224,2)
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowCrouch",1,"Crouch")
			"####" O 4 A_Pain
			Goto See
			
		//Crouching state
		Crouch:
			"####" T 0 A_UnsetShootable
			"####" T 0 A_ChangeFlag("NODAMAGE",true)
			"####" T 0 A_JumpIfInventory("Outlaws_Enemy_CrouchType",2,"Crouch2")
		Crouch1:
			"####" T 16
			Goto UnCrouch
		Crouch2:
			"####" TUTU 4
			Goto UnCrouch
		UnCrouch:
			"####" T 0 A_SetShootable
			"####" T 0 A_ChangeFlag("NODAMAGE",false)
			Goto See
		
		//Death States. May need to be replaced by some enemies
		Death:
			"####" "#" 0 A_Jump(256,"Death1","Death2","Death3")
		Death1:
			"####" ABCD 4
			"####" E -1
			Stop
		Death2:
			"####" FGHIJ 4
			"####" K -1
			Stop
		Death3:
			"####" LMNOP 4
			"####" Q -1
			Stop
	}
}

Actor Outlaws_EnemyPrimary : Trigger
{
	States
	{
		Pickup:
			TNT1 A 0 A_JumpIfInventory("Outlaws_Enemy_ID",1,"AttackSkinnyGuy")
			Stop
		AttackSkinnyGuy:
			TNT1 A 0 A_FaceTarget
			TNT1 A 0 A_PlaySound("Outlaws/Pistol/Fire",CHAN_WEAPON)
			TNT1 A 0 A_CustomBulletAttack(10,10,1,10,"OutlawsPuff")
			Stop
	}
}

Actor Outlaws_Enemy_TauntDelay : TemporaryCounter { Inventory.MaxAmount 20 }
Actor Outlaws_PlayTaunt : Trigger
{
	States
	{
		Pickup:
			TNT1 A 0 A_JumpIfInventory("Outlaws_Enemy_TauntDelay",1,"UseFail")
			TNT1 A 0 A_GiveInventory("Outlaws_Enemy_TauntDelay",20)
			TNT1 A 0 A_Jump(224,"UseFail")
			TNT1 A 0 A_JumpIfTargetInLOS("InLOS",0,JLOSF_DEADNOJUMP)
			Goto NOLOS
		NOLOS:
			TNT1 A 0 A_JumpIfInventory("Outlaws_Enemy_ID",1,"NOLOSSkinnyGuy1")
		NOLOSSkinnyGuy1:
			TNT1 A 0 A_PlaySound("Outlaws/BadGuy1/NOLOS",CHAN_VOICE)
			Stop
		InLOS:
			TNT1 A 0 A_JumpIfInventory("Outlaws_Enemy_ID",1,"LOSSkinnyGuy1")
		LOSSkinnyGuy1:
			TNT1 A 0 A_PlaySound("Outlaws/BadGuy1/LOS",CHAN_VOICE)
			Stop
		UseFail:
			TNT1 A 0 A_TakeInventory("Outlaws_Enemy_TauntDelay",1)
			Stop
	}
}
