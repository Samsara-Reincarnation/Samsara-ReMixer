/*Reserved frames

Standing = A
Walking = B-E
Running = F-I
Shooting = J-N
Pain = O
Melee = P-R
Dynamite = S-T
Crouching = U-V
StrafeRun = W-X
StrafeWalk = Y-Z
StrafeRunType2 = N,P-R
StrafeWalkType2 = W-Z

Death Animations typically have 5 for the first 5, and 6 each for the last 2.
We typically put those in a separate sprite name.

*/

//Generic variables used by this controller
Actor Outlaws_Enemy_AllowWalk : TemporaryBoolean {}
Actor Outlaws_Enemy_AllowRun : TemporaryBoolean {}
Actor Outlaws_Enemy_AllowMelee : TemporaryBoolean {}
Actor Outlaws_Enemy_StrafeBehavior : TemporaryCounter {}
Actor Outlaws_Enemy_ShootType : TemporaryCounter {}
Actor Outlaws_Enemy_AllowDynamite : TemporaryBoolean {}
Actor Outlaws_Enemy_CrouchType : TemporaryCounter {}
Actor Outlaws_Enemy_Difficulty : TemporaryCounter {}
Actor Outlaws_Enemy_ID : TemporaryCounter {}

//Controller definition
Actor OutlawsEnemyCore : SamsaraMonsterHumanoid
{
	Height 56
	Radius 20
	Health 5
	Mass 30
	XScale 0.67
	YScale 0.55833
	Speed 6
	Monster
	Painchance 128
	var int user_speed;
	States
	{
		//Idle/standing
		Spawn:
			"####" A 4 A_Look
			Wait
			
		//Movement based states
		See:
			"####" A 0 A_JumpIfInventory("Outlaws_Enemy_AllowWalk",1,"AllowWalk")
			"####" A 0 A_JumpIfInventory("Outlaws_Enemy_AllowRun",1,"Run")
			Goto StandingEnemy
		StandingEnemy:
			"####" A 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" AAAA 4 A_Chase("Melee","Missile",CHF_DONTMOVE)
			Loop
		AllowWalk:
			"####" "#" 0 A_Jump(64,"Walk")
		Run:
			"####" F 0 ACS_NamedExecuteWithResult("Outlaws_EnemySpeed",user_speed*2)
			"####" F 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" F 4 A_Chase
			"####" G 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" G 4 A_Chase
			"####" H 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" H 4 A_Chase
			"####" I 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" I 4 A_Chase
			Goto See
		Walk:
			"####" F 0 ACS_NamedExecuteWithResult("Outlaws_EnemySpeed",user_speed)
			"####" B 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" B 4 A_Chase
			"####" C 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" C 4 A_Chase
			"####" D 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" D 4 A_Chase
			"####" E 0 A_GiveInventory("Outlaws_PlayTaunt")
			"####" E 4 A_Chase
			Goto See
			
		//Attack based states
		Missile:
			"####" "#" 0 A_Jump(224,2)
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowDynamite",1,"Dynamite")
			"####" "#" 0 A_Jump(224,2)
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_StrafeBehavior",1,"DoStrafe")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",6,"Fire2FramesDual")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",5,"Fire5Frames")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",4,"Fire4Frames")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",3,"Fire3Frames")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_ShootType",2,"Fire2Frames")
		Fire2Frames:
			"####" J 4 Bright A_FaceTarget
			"####" K 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			Goto See
		Fire3Frames:
			"####" JK 4 Bright A_FaceTarget
			"####" L 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			Goto See
		Fire4Frames:
			"####" JK 4 Bright A_FaceTarget
			"####" LM 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			Goto See
		Fire5Frames:
			"####" JKLM 4 Bright A_FaceTarget
			"####" N 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			Goto See
		Fire2FramesDual:
			"####" J 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			"####" K 4 Bright A_GiveInventory("Outlaws_EnemyPrimary")
			Goto See
		Melee:
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowMelee",1,1)
			Goto Missile
			"####" PQR 4 A_FaceTarget
			"####" R 4 A_GiveInventory("Outlaws_EnemyMelee")
			Goto See
		Dynamite:
			"####" S 12 Bright A_GiveInventory("Outlaws_EnemyDynamite")
			Goto See
		DoStrafe:
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_StrafeBehavior",2,"DoStrafe2")
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowWalk",1,1)
			Goto StrafeRun
			"####" "#" 0 A_Jump(256,"StrafeRun","StrafeWalk")
		StrafeRun:
			"####" W 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" W 4 Bright A_ChangeVelocity((user_speed*1.0)/3,0,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" X 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" X 4 Bright A_ChangeVelocity(0,(user_speed*1.0)/3,velz,CVF_RELATIVE|CVF_REPLACE)
			Goto See
		StrafeWalk:
			"####" Y 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" Y 4 Bright A_ChangeVelocity((user_speed*1.0)/6,0,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" Z 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" Z 4 Bright A_ChangeVelocity(0,(user_speed*1.0)/6,velz,CVF_RELATIVE|CVF_REPLACE)
			Goto See
		DoStrafe2:
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowWalk",1,1)
			Goto StrafeRun2
			"####" "#" 0 A_Jump(256,"StrafeRun2","StrafeWalk2")
		StrafeRun2:
			"####" N 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" N 4 Bright A_ChangeVelocity((user_speed*1.0)/3,0,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" P 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" P 4 Bright A_ChangeVelocity((user_speed*1.0)/3,0,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" Q 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" Q 4 Bright A_ChangeVelocity(0,(user_speed*1.0)/3,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" R 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" R 4 Bright A_ChangeVelocity(0,(user_speed*1.0)/3,velz,CVF_RELATIVE|CVF_REPLACE)
			Goto See
		StrafeWalk2:
			"####" W 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" W 4 Bright A_ChangeVelocity((user_speed*1.0)/6,0,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" X 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" X 4 Bright A_ChangeVelocity((user_speed*1.0)/6,0,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" Y 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" Y 4 Bright A_ChangeVelocity(0,(user_speed*1.0)/6,velz,CVF_RELATIVE|CVF_REPLACE)
			"####" Z 0 A_GiveInventory("Outlaws_EnemyPrimary")
			"####" Z 4 Bright A_ChangeVelocity(0,(user_speed*1.0)/6,velz,CVF_RELATIVE|CVF_REPLACE)
			Goto See
			
		//Pain state
		Pain:
			"####" "#" 0 A_Jump(224,2)
			"####" "#" 0 A_JumpIfInventory("Outlaws_Enemy_AllowCrouch",1,"Crouch")
			"####" O 4 A_Pain
			Goto See
			
		//Crouching state
		Crouch:
			"####" T 0 A_UnsetShootable
			"####" T 0 A_ChangeFlag("NODAMAGE",true)
			"####" T 0 A_JumpIfInventory("Outlaws_Enemy_CrouchType",2,"Crouch2")
		Crouch1:
			"####" T 16
			Goto UnCrouch
		Crouch2:
			"####" TUTU 4
			Goto UnCrouch
		UnCrouch:
			"####" T 0 A_SetShootable
			"####" T 0 A_ChangeFlag("NODAMAGE",false)
			Goto See
		
		//Death States. May need to be replaced by some enemies
		Death:
			"####" "#" 0 A_Jump(256,"Death1","Death2","Death3")
		Death1:
			"####" ABCD 4
			"####" E -1
			Stop
		Death2:
			"####" FGHIJ 4
			"####" K -1
			Stop
		Death3:
			"####" LMNOP 4
			"####" Q -1
			Stop
	}
}

Actor Outlaws_EnemyDynamite : Trigger
{
	States
	{
		Pickup:
			TNT1 A 0 A_FaceTarget
			TNT1 A 0 A_CustomMissile("Outlaws_EnemyDynamiteProjectile",0,0,random(-10,10),CMF_AIMDIRECTION,random(-30,-10))
			Stop
	}
}

actor Outlaws_EnemyDynamiteProjectile : Outlaws_DynamiteProjectileTimed
{
    States
    {
      Spawn:
        OL00 A 0 NoDelay A_SetArg(0,random(10,20))
        OL00 A 0 A_ChangeVelocity(sqrt(velx*velx+vely*vely)*((args[0]*1.0)/10),0,velz*((args[0]*1.0)/10),CVF_RELATIVE|CVF_REPLACE)
        OL00 A 0 A_SetAngle(angle+frandom(-10,10))
        OL00 A 0 ACS_NamedExecuteWithResult("Outlaws_GenerateHitBox")
        OL00 A 0 A_GiveInventory("Outlaws_DynamiteProjectileTimer",175)
        OL00 A 0 A_GiveInventory("Outlaws_DynamiteExists")
        OL00 A 0 A_PlaySound("Outlaws/Dynamite/Fuse",CHAN_WEAPON,1,true)
        OL00 A 0 A_ChangeFlag("NOTIMEFREEZE",false)
		Goto SpawnLoop
    }
}
